Topic 01
📦 Array Creation & Declaration
Different ways to create arrays in JavaScript

METHOD 1: ARRAY LITERAL (MOST COMMON)

// Empty array
let arr = [];

// Array with values
let numbers = [1, 2, 3, 4, 5];
let fruits = ["apple", "banana", "orange"];

// Mixed types (not recommended but possible)
let mixed = [1, "hello", true, null, {name: "Ravi"}];

METHOD 2: ARRAY CONSTRUCTOR

// Empty array
let arr = new Array();

// Array with size (creates empty slots)
let arr = new Array(5);  // [empty × 5]

// Array with values
let arr = new Array(1, 2, 3);  // [1, 2, 3]

METHOD 3: Array.of()

// Creates array from arguments
let arr = Array.of(1, 2, 3);  // [1, 2, 3]
let arr = Array.of(5);        // [5] (unlike new Array(5))

METHOD 4: Array.from()

// From string
let arr = Array.from("hello");  // ['h','e','l','l','o']

// From range (1 to 5)
let arr = Array.from({length: 5}, (_, i) => i + 1);  // [1,2,3,4,5]

// From Set
let set = new Set([1, 2, 2, 3]);
let arr = Array.from(set);  // [1, 2, 3]
Best Practice: Use [] (array literal) for most cases. It's clearest and most efficient.

Topic 02
🔍 Accessing Array Elements
How to read and modify array elements

BASIC ACCESS WITH INDEX

let fruits = ["apple", "banana", "orange", "mango"];

// Access by index (starts at 0)
console.log(fruits[0]);  // "apple"
console.log(fruits[2]);  // "orange"

// Last element
console.log(fruits[fruits.length - 1]);  // "mango"

// Negative indices don't work (unlike Python)
console.log(fruits[-1]);  // undefined (NOT "mango")

MODIFY ELEMENTS

let nums = [10, 20, 30];

// Change value at index
nums[1] = 99;
console.log(nums);  // [10, 99, 30]

// Add element beyond length (creates gaps)
nums[5] = 50;
console.log(nums);  // [10, 99, 30, empty × 2, 50]

ARRAY PROPERTIES

let arr = [1, 2, 3, 4, 5];

// length — number of elements
console.log(arr.length);  // 5

// First element
console.log(arr[0]);  // 1

// Last element (using length)
console.log(arr[arr.length - 1]);  // 5

at() METHOD (Modern ES2022)

let fruits = ["apple", "banana", "orange"];

// Positive index
console.log(fruits.at(0));   // "apple"

// Negative index (counts from end!)
console.log(fruits.at(-1));  // "orange" ✅
console.log(fruits.at(-2));  // "banana"

Topic 03
🔧 Basic Array Methods
Essential methods for adding, removing, and manipulating arrays
push()
Add element(s) to END. Returns new length.
let arr = [1, 2];
arr.push(3); // [1, 2, 3]
arr.push(4, 5); // [1,2,3,4,5]
pop()
Remove last element. Returns removed element.
let arr = [1, 2, 3];
let last = arr.pop();
// arr = [1, 2], last = 3
unshift()
Add element(s) to START. Returns new length.
let arr = [2, 3];
arr.unshift(1);
// arr = [1, 2, 3]
shift()
Remove first element. Returns removed element.
let arr = [1, 2, 3];
let first = arr.shift();
// arr = [2, 3], first = 1
splice()
Add/remove elements anywhere. Modifies original array.
let arr = [1,2,3,4,5];
arr.splice(2, 1, 99);
// [1,2,99,4,5]
// (start, deleteCount, items...)
slice()
Extract portion. Returns NEW array. Original unchanged.
let arr = [1,2,3,4,5];
let sub = arr.slice(1, 3);
// sub = [2, 3]
// arr still [1,2,3,4,5]

SPLICE() — DETAILED EXAMPLES

let arr = [1, 2, 3, 4, 5];

// Remove 2 elements starting at index 1
arr.splice(1, 2);  // arr = [1, 4, 5]

// Insert elements at index 2 (delete 0)
arr = [1, 2, 3, 4, 5];
arr.splice(2, 0, 99, 100);  // [1, 2, 99, 100, 3, 4, 5]

// Replace elements (delete 2, add 2)
arr = [1, 2, 3, 4, 5];
arr.splice(1, 2, 'a', 'b');  // [1, 'a', 'b', 4, 5]

// Remove everything from index 2 onwards
arr = [1, 2, 3, 4, 5];
arr.splice(2);  // arr = [1, 2]

COMPARISON TABLE

Method Adds Where? Removes From? Modifies Original? Returns
push() End — Yes New length
pop() — End Yes Removed element
unshift() Start — Yes New length
shift() — Start Yes Removed element
splice() Anywhere Anywhere Yes Removed elements
slice() — — No New array

Topic 04
🔎 Search & Find Methods
Methods to search for elements in arrays
indexOf()
Find first index of element. Returns -1 if not found.
let arr = [1,2,3,2,1];
arr.indexOf(2); // 1
arr.indexOf(99); // -1
lastIndexOf()
Find last index of element. Returns -1 if not found.
let arr = [1,2,3,2,1];
arr.lastIndexOf(2); // 3
arr.lastIndexOf(99); // -1
includes()
Check if element exists. Returns true/false.
let arr = [1, 2, 3];
arr.includes(2); // true
arr.includes(99); // false
find()
Find first element matching condition. Returns element or undefined.
let arr = [5, 12, 8, 130];
arr.find(x => x > 10);
// 12 (first > 10)
findIndex()
Find index of first match. Returns -1 if not found.
let arr = [5, 12, 8, 130];
arr.findIndex(x => x > 10);
// 1 (index of 12)
findLast()
Find LAST element matching condition. (ES2023)
let arr = [5, 12, 8, 130];
arr.findLast(x => x > 10);
// 130 (last > 10)
findLastIndex()
Find index of LAST match. (ES2023)
let arr = [5, 12, 8, 130];
arr.findLastIndex(x > 10);
// 3 (index of 130)

DETAILED EXAMPLES

let numbers = [5, 12, 8, 130, 44];

// indexOf() — simple search
numbers.indexOf(8);  // 2
numbers.indexOf(99); // -1 (not found)

// includes() — existence check
numbers.includes(12);  // true
numbers.includes(99);  // false

// find() — with condition
let result = numbers.find(num => num > 10);
console.log(result);  // 12 (first element > 10)

// findIndex() — get index with condition
let index = numbers.findIndex(num => num > 100);
console.log(index);  // 3 (index of 130)

// Find in array of objects
let users = [
  { id: 1, name: "Ravi" },
  { id: 2, name: "Priya" },
  { id: 3, name: "Amit" }
];

let user = users.find(u => u.id === 2);
console.log(user);  // { id: 2, name: "Priya" }

WHEN TO USE WHICH?

Use Case Method Example
Check if value exists includes() arr.includes(5)
Get index of value indexOf() arr.indexOf(5)
Find with complex condition find() arr.find(x => x > 10)
Get index with condition findIndex() arr.findIndex(x => x > 10)
Find object in array find() users.find(u => u.id === 2)

🔁 Looping Through Arrays
Different ways to iterate over array elements

1. TRADITIONAL for LOOP

let fruits = ["apple", "banana", "orange"];

for (let i = 0; i < fruits.length; i++) {
  console.log(fruits[i]);
}
// Output: apple, banana, orange

2. forEach() METHOD

let fruits = ["apple", "banana", "orange"];

// forEach(element, index, array)
fruits.forEach(function(fruit) {
  console.log(fruit);
});

// With arrow function
fruits.forEach(fruit => console.log(fruit));

// With index
fruits.forEach((fruit, index) => {
  console.log(`${index}: ${fruit}`);
});
// 0: apple
// 1: banana
// 2: orange

3. for...of LOOP

let fruits = ["apple", "banana", "orange"];

for (let fruit of fruits) {
  console.log(fruit);
}
// Output: apple, banana, orange

// With entries() to get index too
for (let [index, fruit] of fruits.entries()) {
  console.log(`${index}: ${fruit}`);
}

4. for...in LOOP (NOT RECOMMENDED FOR ARRAYS)

let fruits = ["apple", "banana", "orange"];

// ⚠️ for...in gives indices (strings), not values
for (let index in fruits) {
  console.log(index, typeof index);
}
// 0 string
// 1 string
// 2 string

// Use for...of instead!

5. WHILE LOOP

let fruits = ["apple", "banana", "orange"];
let i = 0;

while (i < fruits.length) {
  console.log(fruits[i]);
  i++;
}

COMPARISON TABLE

Method Access Index? Can break? Can continue? Best For
for loop ✅ Yes ✅ Yes ✅ Yes Full control, complex logic
forEach() ✅ Yes (2nd param) ❌ No ❌ No Simple iteration
for...of ❌ No (unless with entries()) ✅ Yes ✅ Yes Clean iteration of values
for...in ✅ Yes (but as string) ✅ Yes ✅ Yes Objects (avoid for arrays)

Topic 06
🔄 Transform Methods (map, filter, reduce)
Powerful methods to transform and process arrays

map() — TRANSFORM EACH ELEMENT

Creates a NEW array by transforming every element. Original array unchanged.

// Syntax: arr.map((element, index, array) => { })

let numbers = [1, 2, 3, 4, 5];

// Double each number
let doubled = numbers.map(num => num * 2);
console.log(doubled);  // [2, 4, 6, 8, 10]

// Convert to strings
let strings = numbers.map(num => `Number ${num}`);
console.log(strings);
// ["Number 1", "Number 2", "Number 3", "Number 4", "Number 5"]

// Extract property from objects
let users = [
  { id: 1, name: "Ravi" },
  { id: 2, name: "Priya" }
];
let names = users.map(user => user.name);
console.log(names);  // ["Ravi", "Priya"]

filter() — SELECT ELEMENTS BY CONDITION

Creates a NEW array with elements that pass the test. Original unchanged.

// Syntax: arr.filter((element, index, array) => { })

let numbers = [1, 2, 3, 4, 5, 6, 7, 8];

// Get only even numbers
let evens = numbers.filter(num => num % 2 === 0);
console.log(evens);  // [2, 4, 6, 8]

// Get numbers > 5
let large = numbers.filter(num => num > 5);
console.log(large);  // [6, 7, 8]

// Filter objects
let users = [
  { name: "Ravi", age: 25 },
  { name: "Priya", age: 30 },
  { name: "Amit", age: 20 }
];
let adults = users.filter(user => user.age >= 25);
console.log(adults);
// [{ name: "Ravi", age: 25 }, { name: "Priya", age: 30 }]

reduce() — REDUCE TO SINGLE VALUE

Reduces array to a single value by applying a function to each element.

// Syntax: arr.reduce((accumulator, current, index, array) => { }, initialValue)

let numbers = [1, 2, 3, 4, 5];

// Sum all numbers
let sum = numbers.reduce((total, num) => total + num, 0);
console.log(sum);  // 15

// Find maximum
let max = numbers.reduce((max, num) => num > max ? num : max, numbers[0]);
console.log(max);  // 5

// Count occurrences
let fruits = ["apple", "banana", "apple", "orange", "banana", "apple"];
let count = fruits.reduce((acc, fruit) => {
  acc[fruit] = (acc[fruit] || 0) + 1;
  return acc;
}, {});
console.log(count);
// { apple: 3, banana: 2, orange: 1 }

// Flatten array
let nested = [[1, 2], [3, 4], [5]];
let flat = nested.reduce((acc, arr) => acc.concat(arr), []);
console.log(flat);  // [1, 2, 3, 4, 5]

CHAINING METHODS

// Combine map(), filter(), reduce()
let numbers = [1, 2, 3, 4, 5, 6, 7, 8];

// Get sum of squares of even numbers
let result = numbers
  .filter(num => num % 2 === 0)     // [2, 4, 6, 8]
  .map(num => num * num)          // [4, 16, 36, 64]
  .reduce((sum, num) => sum + num, 0);  // 120

console.log(result);  // 120

COMPARISON TABLE

Method Purpose Returns Changes Original?
map() Transform each element New array (same length) No
filter() Select elements by condition New array (≤ original length) No
reduce() Reduce to single value Single value (any type) No

Topic 07
🔢 Array Sorting
Different ways to sort arrays in JavaScript

sort() — BASIC SORTING

⚠️ WARNING: sort() modifies the ORIGINAL array and converts elements to strings by default!

// String sorting (works as expected)
let fruits = ["banana", "apple", "orange"];
fruits.sort();
console.log(fruits);  // ["apple", "banana", "orange"]

// ⚠️ Number sorting (WRONG without compare function!)
let numbers = [10, 5, 40, 25, 1000];
numbers.sort();
console.log(numbers);  // [10, 1000, 25, 40, 5] ❌ WRONG!
// Why? It converts to strings: "10" < "1000" < "25"...

SORTING NUMBERS CORRECTLY

let numbers = [10, 5, 40, 25, 1000];

// Ascending order (smallest first)
numbers.sort((a, b) => a - b);
console.log(numbers);  // [5, 10, 25, 40, 1000] ✅

// Descending order (largest first)
numbers.sort((a, b) => b - a);
console.log(numbers);  // [1000, 40, 25, 10, 5] ✅

// How it works:
// If return < 0: a comes before b
// If return > 0: b comes before a
// If return = 0: order unchanged

SORTING OBJECTS

let users = [
  { name: "Ravi", age: 25 },
  { name: "Amit", age: 30 },
  { name: "Priya", age: 20 }
];

// Sort by age (ascending)
users.sort((a, b) => a.age - b.age);
console.log(users);
// [{ name: "Priya", age: 20 }, { name: "Ravi", age: 25 }, { name: "Amit", age: 30 }]

// Sort by name (alphabetically)
users.sort((a, b) => a.name.localeCompare(b.name));
console.log(users);
// [{ name: "Amit", ... }, { name: "Priya", ... }, { name: "Ravi", ... }]

reverse() — REVERSE ORDER

let arr = [1, 2, 3, 4, 5];

// Reverse in place (modifies original)
arr.reverse();
console.log(arr);  // [5, 4, 3, 2, 1]

// Reverse without modifying original
let original = [1, 2, 3];
let reversed = [...original].reverse();
console.log(original);  // [1, 2, 3] (unchanged)
console.log(reversed);  // [3, 2, 1]

toSorted() & toReversed() — NON-MUTATING (ES2023)

// New methods that DON'T modify original array
let numbers = [3, 1, 4, 1, 5];

// toSorted() — returns new sorted array
let sorted = numbers.toSorted((a, b) => a - b);
console.log(numbers);  // [3, 1, 4, 1, 5] (unchanged)
console.log(sorted);   // [1, 1, 3, 4, 5]

// toReversed() — returns new reversed array
let reversed = numbers.toReversed();
console.log(numbers);   // [3, 1, 4, 1, 5] (unchanged)
console.log(reversed);  // [5, 1, 4, 1, 3]

ADVANCED SORTING EXAMPLES

// Case-insensitive string sorting
let names = ["charlie", "Alice", "bob"];
names.sort((a, b) => a.toLowerCase().localeCompare(b.toLowerCase()));
console.log(names);  // ["Alice", "bob", "charlie"]

// Multi-level sorting (age, then name)
let people = [
  { name: "Ravi", age: 25 },
  { name: "Amit", age: 25 },
  { name: "Priya", age: 20 }
];

people.sort((a, b) => {
  if (a.age !== b.age) {
    return a.age - b.age;  // Sort by age first
  }
  return a.name.localeCompare(b.name);  // Then by name
});

// Random shuffle
let arr = [1, 2, 3, 4, 5];
arr.sort(() => Math.random() - 0.5);
console.log(arr);  // Random order
