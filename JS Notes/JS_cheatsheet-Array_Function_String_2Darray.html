TRANSFORM METHODS — DEEP DIVE

let nums = [1,2,3,4,5];

// map() — transform each
nums.map(n => n * 2);          // [2,4,6,8,10]

// filter() — select by condition
nums.filter(n => n % 2 === 0);  // [2,4]

// reduce() — single value
nums.reduce((acc,n) => acc + n, 0); // 15

// some() — any match?
nums.some(n => n > 4);          // true

// every() — all match?
nums.every(n => n > 0);          // true

// flat() — flatten nested
[1,[2,[3,[4]]]].flat();          // [1,2,[3,[4]]]
[1,[2,[3,[4]]]].flat(Infinity); // [1,2,3,4]

// flatMap() — map + flat(1)
[1,2,3].flatMap(n => [n, n * 2]); // [1,2,2,4,3,6]

// CHAINING
nums
  .filter(n => n % 2 === 0)  // [2,4]
  .map(n => n * n)              // [4,16]
  .reduce((a,b) => a + b, 0);  // 20

let nums = [10,5,40,25];
let strs = ['banana','apple','cherry'];

// Numbers ascending
nums.sort((a,b) => a - b);              // [5,10,25,40]

// Numbers descending
nums.sort((a,b) => b - a);              // [40,25,10,5]

// Strings alphabetical
strs.sort();                             // ['apple','banana','cherry']

// Strings reverse alpha
strs.sort((a,b) => b.localeCompare(a)); // ['cherry','banana','apple']

// Objects by property
let users = [{name:'Bob',age:30},{name:'Alice',age:25}];

// Shuffle randomly
nums.sort(() => Math.random() - 0.5);

FUNCTION DECLARATION

function greet(name) {
  return `Hello, ${name}!`;
}

ARROW FUNCTION

  return a + b;

DEFAULT PARAMETERS

  return `${host}:${port}`;
}

  return nums.reduce((a,b) => a + b, 0);

sum(1, 2, 3, 4, 5); // 15
// ...nums collects all args into array

HIGHER ORDER FUNCTION

  return (num) => num * factor;

CLOSURE

function counter() {
  let count = 0;   // private variable
  return {
    increment: () => ++count,
    decrement: () => --count,
    value: () => count
  };

c.increment();  // 1
c.increment();  // 2
c.value();      // 2

PROBLEM 2: Two Sum (Google/Amazon)

Find two indices that add up to target. Input: [2,7,11,15], target: 9

function twoSum(arr, target) {
  let map = new Map();
  for (let i = 0; i < arr.length; i++) {
    let complement = target - arr[i];
    if (map.has(complement)) {
      return [map.get(complement), i];
    }
    map.set(arr[i], i);
  }
}
twoSum([2,7,11,15], 9); // [0,1]

PROBLEM 4: Flatten Deep Nested Array

Flatten [1,[2,[3,[4,[5]]]]] completely.

// Method 1: flat(Infinity)

  return arr.reduce((acc, val) =>
    Array.isArray(val) ? acc.concat(flatten(val)) : [...acc, val], []);

PROBLEM 10: Most Frequent Character

function mostFrequent(str) {
  let freq = {};
  for (let ch of str) freq[ch] = (freq[ch] || 0) + 1;
  return Object.keys(freq).reduce((a,b) => freq[a] > freq[b] ? a : b);
}
mostFrequent('javascript'); // 'a'

PROBLEM 13: Memoization / Caching

function memoize(fn) {
  let cache = {};
  return function(...args) {
    let key = JSON.stringify(args);
    if (key in cache) return cache[key];
    cache[key] = fn(...args);
    return cache[key];
  };
}
const factorial = memoize(n => n <= 1 ? 1 : n * factorial(n-1));
factorial(5); // 120

PROBLEM 15: Second Largest (Amazon)

function secondLargest(arr) {
  let first = -Infinity, second = -Infinity;
  for (let n of arr) {
    if (n > first) { second = first; first = n; }
    else if (n > second && n < first) second = n;
  }
  return second === -Infinity ? null : second;
}
secondLargest([45,78,23,90,56,12,90]); // 78

🎯 Interview Questions & Answers
Top questions asked at FAANG, product and service companies

📦 ARRAY INTERVIEW QUESTIONS

Q: Difference between map() and forEach()?
map() returns a NEW array with transformed values. forEach() returns undefined — used only for side effects. Use map() when you need a new array, forEach() when you just need to loop.
Q: Can you break out of forEach?
No. forEach cannot be stopped with break or continue. Use for...of with break, or use some() (return true to stop) or every() (return false to stop).
Q: Difference between splice() and slice()?
splice() modifies the ORIGINAL array (add/remove). slice() creates a NEW array (extract). Memory trick: splIce = In-place (mutates). slIce = Independent copy.
Q: Shallow vs deep copy of arrays?
[...arr] and slice() are shallow — nested objects share reference. Use structuredClone() or JSON.parse(JSON.stringify()) for deep copy. Spread copies only one level deep.
Q: What does reduce() do?
Reduces an array to a single value by accumulating a result through each element. Can produce numbers, strings, objects, or arrays. The accumulator holds the running result.
Q: Difference between find() and filter()?
find() returns the FIRST matching element (or undefined). filter() returns ALL matching elements as a new array. Use find() when you want one result, filter() for multiple.

📝 STRING INTERVIEW QUESTIONS

Q: Are strings mutable in JS?
No. Strings are immutable. Methods like toUpperCase(), slice(), replace() return NEW strings — they never modify the original. You can only replace the variable's value entirely.
Q: Difference between slice() and substring()?
Both extract substrings. slice() supports negative indices (counts from end). substring() treats negatives as 0 and swaps args if start > end. slice() is more powerful and modern.
Q: How to check if string contains a substring?
Use includes() for boolean check. indexOf() for position (-1 if not found). match() or search() for regex patterns. startsWith()/endsWith() for beginning/end checks.
Q: Template literals vs string concatenation?
Template literals (backticks) are cleaner and support multi-line strings and expressions. `Hello ${name}` vs "Hello " + name. Template literals are the modern standard.

🗂️ OBJECT INTERVIEW QUESTIONS

Q: Difference between Object.freeze() and Object.seal()?
freeze() — cannot add, delete, or modify any property. seal() — cannot add or delete, but CAN modify existing values. freeze is stricter than seal.
Q: What is destructuring?
A shorthand to extract values from arrays/objects into variables. let {name, age} = user; instead of let name = user.name. Supports defaults, renaming, and rest (...) syntax.
Q: Difference between null and undefined?
undefined — variable declared but not assigned. null — intentional absence of value (explicitly set). typeof undefined is "undefined", typeof null is "object" (historical bug).
Q: What is optional chaining (?.)?
Safely access deeply nested properties without errors. user?.address?.city returns undefined instead of throwing if address is null/undefined. Very useful for API responses.

⚡ FUNCTION INTERVIEW QUESTIONS

Q: What is a closure?
A function that remembers variables from its outer scope even after the outer function has finished. Used for data privacy, memoization, and factory functions. The inner function "closes over" outer variables.
Q: Arrow functions vs regular functions?
Arrow functions don't have their own 'this' — they inherit from surrounding scope. Regular functions have their own 'this'. Arrows can't be used as constructors. No 'arguments' object in arrows.
Q: What is hoisting?
Function declarations are hoisted (can call before defining). var is hoisted as undefined. let/const are hoisted but in TDZ (Temporal Dead Zone) — accessing before declaration throws ReferenceError.
Q: What is a callback function?
A function passed as an argument to another function, called when a task completes. Used in array methods (map, filter, forEach), event handlers, and async operations (setTimeout, fetch).
